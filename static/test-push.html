🔧 Setup Instructions

Firebase Configuration

  1. Go to Firebase Console
  2. Create a new project or select existing one
  3. Go to Project Settings → Service Accounts
  4. Click Generate New Private Key
  5. Save the JSON file as config/firebase-credentials.json in your project
  6. Add iOS and/or Android apps in Firebase project settings
  7. Download the config files:
    • iOS: GoogleService-Info.plist
    • Android: google-services.json
  8. Restart the API server

iOS Configuration

  1. Add GoogleService-Info.plist to your iOS project
  2. Install Firebase SDK via CocoaPods: 
    pod 'Firebase/Messaging'
  3. Initialize Firebase in AppDelegate:
    4. import Firebase
import FirebaseMessaging

func application(_ application: UIApplication, 
                 didFinishLaunchingWithOptions launchOptions: [UIApplication.LaunchOptionsKey: Any]?) -> Bool {
    FirebaseApp.configure()
    Messaging.messaging().delegate = self
    UNUserNotificationCenter.current().delegate = self
    
    let authOptions: UNAuthorizationOptions = [.alert, .badge, .sound]
    UNUserNotificationCenter.current().requestAuthorization(options: authOptions) { granted, _ in
        print("Permission granted: \(granted)")
    }
    application.registerForRemoteNotifications()
    return true
}
  4. Get FCM token:
    5. Messaging.messaging().token { token, error in
    if let token = token {
        print("FCM Token: \(token)")
        // Send this token to your server
    }
}
  5. Enable Push Notifications capability in Xcode
  6. Upload APNs certificate to Firebase Console

Android Configuration

  1. Add google-services.json to app/ directory
  2. Add dependencies to build.gradle:
    3. dependencies {
    implementation platform('com.google.firebase:firebase-bom:32.7.0')
    implementation 'com.google.firebase:firebase-messaging'
}
  3. Initialize Firebase in MainActivity:
    4. import com.google.firebase.messaging.FirebaseMessaging

FirebaseMessaging.getInstance().token.addOnCompleteListener { task ->
    if (task.isSuccessful) {
        val token = task.result
        Log.d("FCM", "Token: $token")
        // Send this token to your server
    }
}
  4. Create FirebaseMessagingService:
    5. class MyFirebaseMessagingService : FirebaseMessagingService() {
    override fun onMessageReceived(remoteMessage: RemoteMessage) {
        remoteMessage.notification?.let {
            Log.d("FCM", "Message: ${it.body}")
        }
    }
    
    override fun onNewToken(token: String) {
        Log.d("FCM", "New token: $token")
    }
}
